\section{Réciproque du théorème de Pythagore}
\subsection{Enoncé de la réciproque}
\begin{theorem}[(Réciproque du théorème de Pythagore)]
    Soit NRV un triangle tel que
    $$NR^2=RV^2+NV^2$$
    Alors, \bb{d'après la réciproque du théorème de Pythagore}, le triangle NRV
    est un \bb{triangle rectangle en V}.
\end{theorem}

\subsection{Application : Montrer qu'un triangle est rectangle}
\begin{exe}
    Soit un triangle $ENT$ tel que 
    $$EN=7.2cm\hspace{2cm}NT=5.4cm\hspace{2cm}ET=9cm$$
    \bb{Le plus grand côté} est $\answer{ET}$. Calculons alors $\answer{ET^2}$ 
    et $\answer{EN^2+NT^2}$ pour vérifier si l'égalité de Pythagore est vérifiée.\\
    \begin{minipage}{0.48\textwidth}
        \begin{align*}
            \answer{ET^2}&=\answer{9^2}\\
            &=\answer{81}\\
        \end{align*}
    \end{minipage}
    \hfill
    \begin{minipage}{0.48\textwidth}
        \begin{align*}
            \answer{EN^2+NT^2}&=\answer{7.2^2+5.4^2}\\
            &=\answer{51.84 + 29.16}\\
            &=\answer{81}\\
        \end{align*}
    \end{minipage}\\
    On a donc bien $\answer{ET^2=EN^2+NT^2}$ donc \bb{$ENT$ est un triangle
    rectangle en T}.
\end{exe}
