\section{Soustraire des nombres relatifs}
\begin{propbox}
Pour soustraire un nombre relatif à un autre, on y ajoute son opposé.
$$a-b=a+(-b)$$
\end{propbox}

\begin{exemplebox}
    $$(+7)-(+3) = \answer{(+7)+(-3) = +4}$$
    $$(+7)-(-3) = \answer{(+7)+(+3) = +10}$$
    $$(-5)-(+4) = \answer{(-5)+(-4) = -9}$$
\end{exemplebox}
