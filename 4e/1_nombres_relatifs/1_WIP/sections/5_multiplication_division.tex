\section{Multiplier et diviser des nombres relatifs}

\begin{propbox}
\bb{Règle des signes}

Pour un produit ou un quotient :

\begin{itemize}
    \item même signe $\rightarrow$ résultat \bb{positif} ;
    \item signes contraires $\rightarrow$ résultat \bb{négatif}.
\end{itemize}
\end{propbox}

\begin{center}
\begin{tabular}{c|c|c}
Signe & Signe & Résultat \\ \hline
$+$ & $+$ & $+$ \\
$-$ & $-$ & $+$ \\
$+$ & $-$ & $-$ \\
$-$ & $+$ & $-$
\end{tabular}
\end{center}

\begin{exemplebox}
\[
(+4)\times(+3)=+12
\]

\[
(-4)\times(-3)=+12
\]

\[
(-4)\times(+3)=-12
\]

\[
(+12)\div(-3)=-4
\]
\end{exemplebox}

\begin{methodebox}
Pour calculer un produit ou un quotient :

\begin{enumerate}
    \item Je détermine le signe du résultat.
    \item Je calcule avec les distances à zéro.
\end{enumerate}
\end{methodebox}

\begin{attentionbox}
\bb{Deux signes identiques donnent un résultat positif.}

\bb{Deux signes différents donnent un résultat négatif.}
\end{attentionbox}
