\newenvironment{tableau}[1]
{%
    \renewcommand{\arraystretch}{1.3}%
    \begin{center}%
    \begin{tabularx}{\linewidth}{#1}%
    \toprule
}
{%
    \bottomrule
    \end{tabularx}%
    \end{center}%
    \renewcommand{\arraystretch}{1}%
}


\newenvironment{exocol}{%
    \vspace{0pt}%
    \begin{minipage}[t]{0.47\textwidth}%
    \vspace{0pt}%
}{%
    \end{minipage}%
}



\newenvironment{exopage}
{%
    \thispagestyle{empty}
}
{%
    \clearpage
}


\newenvironment{exogrid}[1][2]{%
    \begin{tcbraster}[
        raster columns=#1,
        raster column skip=5mm,
        raster row skip=5mm,
        raster valign=top,
        raster every box/.style={
            blankest,
            boxsep=0pt,
            left=0pt,
            right=0pt,
            top=0pt,
            bottom=0pt,
        }
    ]
}{%
    \end{tcbraster}
}
