\section{Calculer avec des puissances}

\begin{definitionbox}
    Quel que soit le nombre x:
    \begin{align*}
        x^2 &= x \times x\\
        x^3 &= x \times x\times x
    \end{align*}
\end{definitionbox}
\begin{vocab}
    \begin{itemize}
        \item $8^2$ se lit "8 au \answer{carré}" ou "8 \answer{puissance} 2" ou
            "8 \answer{exposant} 2" et
            signifie $8 \times 8$
        \item $5^3$ se lit "5 au \answer{cube}" ou "5 \answer{puissance} 3" ou
            "5 \answer{exposant} 3" et
            signifie $5 \times 5\times 5$
    \end{itemize}
\end{vocab}
